-- Таблиця курсів
CREATE TABLE courses (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255) NOT NULL,                  -- назва курсу
    language ENUM('en', 'da') NOT NULL,          -- мова курсу
    start_datetime DATETIME NOT NULL,            -- дата і час початку курсу
    rescheduled_datetime DATETIME DEFAULT NULL,  -- нова дата і час початку (якщо перенесено)
    days_of_week VARCHAR(100) DEFAULT NULL,      -- дні проведення курсу (наприклад: "Mon,Wed,Fri")
    location VARCHAR(255) DEFAULT NULL           -- місце проведення курсу
);

-- Таблиця блогів
CREATE TABLE blog_posts (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    title VARCHAR(255) NOT NULL,                 -- заголовок блогу
    description TEXT DEFAULT NULL,               -- короткий опис
    image_url VARCHAR(255) DEFAULT NULL,         -- шлях або URL до зображення
    content LONGTEXT NOT NULL,                   -- повний текст статті
    category VARCHAR(100) DEFAULT NULL           -- категорія
); 

CREATE TABLE blogs (
  id INT AUTO_INCREMENT PRIMARY KEY,
  title VARCHAR(255),
  content LONGTEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Таблиця користувачів
CREATE TABLE users (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255) NOT NULL,                  -- ім'я користувача
    user_type ENUM('admin','user','editor') NOT NULL, -- тип користувача
    password_hash VARCHAR(255) NOT NULL          -- хеш пароля
);

-- Таблиця контактної інформації
CREATE TABLE contact_info (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    work_hours VARCHAR(255) DEFAULT NULL,        -- робочі години
    phone_time VARCHAR(255) DEFAULT NULL,        -- час, коли можна телефонувати
    phone VARCHAR(50) DEFAULT NULL,              -- номер телефону
    email VARCHAR(255) DEFAULT NULL,             -- електронна пошта
    address VARCHAR(255) DEFAULT NULL            -- адреса
);

CREATE TABLE users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    failed_attempts INT DEFAULT 0,
    locked_until INT DEFAULT NULL,
    created_at INT NOT NULL
);

CREATE TABLE login_attempts (
    id INT AUTO_INCREMENT PRIMARY KEY,
    ip VARCHAR(45) NOT NULL,
    attempt_time INT NOT NULL,
    INDEX(ip),
    INDEX(attempt_time)
);
